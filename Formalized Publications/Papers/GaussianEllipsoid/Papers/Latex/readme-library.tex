\begin{editedaiblock}

\begin{verbatim}
    \begin{editedaiblock}

    \end{editedaiblock}.
\end{verbatim}

\humanul{We can highlight human edits to the AI output with the human underline command}
\begin{verbatim}
    \humanul{We can highlight human edits to the AI output...}
\end{verbatim}

\end{editedaiblock}
